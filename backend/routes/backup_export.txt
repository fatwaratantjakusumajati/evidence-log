// HELPER: CEK PELANGGARAN JAM KERJA
// ============================================================
function hitungPelanggaran(attendanceData) {
  var pelanggaran = 0;

  attendanceData.forEach(function (a) {
    var waktu = new Date(a.timestamp);
    var jam = waktu.getHours();
    var menit = waktu.getMinutes();
    var totalMenit = jam * 60 + menit;

    if (a.event_type === "ARRIVAL") {
      // Telat: masuk setelah 07:00
      if (totalMenit > 7 * 60) {
        pelanggaran++;
      }
    }

    if (a.event_type === "BREAK_OUT") {
      // Cek apakah istirahat di luar waktu yang ditentukan
      // Istirahat 3x: 09:00-09:20 (540-560), 12:00-12:45 (720-765), 15:00-15:20 (900-920)
      // Istirahat 1x: 12:00-13:00 (720-780)
      var isValidBreak = false;

      // Cek apakah dalam rentang istirahat yang valid
      if (totalMenit >= 540 && totalMenit <= 560) isValidBreak = true; // 09:00-09:20
      if (totalMenit >= 720 && totalMenit <= 765) isValidBreak = true; // 12:00-12:45
      if (totalMenit >= 900 && totalMenit <= 920) isValidBreak = true; // 15:00-15:20
      if (totalMenit >= 720 && totalMenit <= 780) isValidBreak = true; // 12:00-13:00 (istirahat 1x)

      if (!isValidBreak) {
        pelanggaran++;
      }
    }

    if (a.event_type === "DEPARTURE") {
      // Pulang sebelum 16:00
      if (totalMenit < 16 * 60) {
        pelanggaran++;
      }
    }
  });

  return pelanggaran;
}

// SVG CHARTS
// ============================================================
function generateBarChart(data, labelKey, valueKey, maxValue, color) {
  const totalData = data.reduce((sum, d) => sum + (d[valueKey] || 0), 0);
  if (!data || data.length === 0 || totalData === 0) {
    return '<div style="text-align:center;padding:30px;color:#94a3b8;font-size:12px;">📊 No data available</div>';
  }

  var width = 340,
    height = 130,
    pad = 30;
  var cw = width - pad * 2,
    ch = height - pad * 2;
  var shown = data.slice(-8);
  var slot = cw / shown.length;
  var bw = Math.min(slot - 10, 28);
  var bars = "",
    labels = "";

  for (var i = 0; i < shown.length; i++) {
    var d = shown[i];
    var val = d[valueKey] || 0;
    var bh = maxValue > 0 ? (val / maxValue) * ch : 0;
    var x = pad + i * slot + (slot - bw) / 2;
    var y = pad + ch - bh;

    bars +=
      '<rect x="' +
      x.toFixed(1) +
      '" y="' +
      y.toFixed(1) +
      '" width="' +
      bw.toFixed(1) +
      '" height="' +
      Math.max(bh, 1).toFixed(1) +
      '" fill="' +
      color +
      '" rx="4" opacity="0.85"/>';
    bars +=
      '<text x="' +
      (x + bw / 2).toFixed(1) +
      '" y="' +
      (y - 6) +
      '" text-anchor="middle" font-size="10" font-weight="bold" fill="' +
      color +
      '">' +
      val +
      "</text>";
    labels +=
      '<text x="' +
      (x + bw / 2).toFixed(1) +
      '" y="' +
      (pad + ch + 16) +
      '" text-anchor="middle" font-size="9" fill="#64748b">' +
      d[labelKey] +
      "</text>";
  }

  return (
    '<svg width="' +
    width +
    '" height="' +
    (height + 20) +
    '" viewBox="0 0 ' +
    width +
    " " +
    (height + 20) +
    '" style="width:100%;height:auto;display:block;">' +
    bars +
    labels +
    "</svg>"
  );
}

function generateDonutChart(total, offline) {
  var online = total - offline;
  var percent = total > 0 ? Math.round((online / total) * 100) : 0;
  var r = 38,
    circ = 2 * Math.PI * r;
  var offset = circ - (percent / 100) * circ;
  var color = offline > 0 ? "#f59e0b" : "#10b981";

  if (total === 0) {
    return '<div style="text-align:center;padding:30px;color:#94a3b8;font-size:12px;">📡 No data</div>';
  }

  return (
    '<div style="display:flex;align-items:center;gap:16px;padding:8px 0;">' +
    '<svg width="110" height="100" viewBox="0 0 110 100">' +
    '<circle cx="55" cy="45" r="' +
    r +
    '" fill="none" stroke="#f1f5f9" stroke-width="10"/>' +
    '<circle cx="55" cy="45" r="' +
    r +
    '" fill="none" stroke="' +
    color +
    '" stroke-width="10" stroke-dasharray="' +
    circ +
    '" stroke-dashoffset="' +
    offset +
    '" transform="rotate(-90 55 45)" stroke-linecap="round"/>' +
    '<text x="55" y="42" text-anchor="middle" font-size="20" font-weight="bold" fill="#1e293b">' +
    total +
    "</text>" +
    '<text x="55" y="58" text-anchor="middle" font-size="9" fill="#64748b">CAMERAS</text>' +
    "</svg>" +
    '<div style="display:flex;flex-direction:column;gap:10px;font-size:11px;">' +
    '<div style="display:flex;align-items:center;gap:8px;"><span style="width:10px;height:10px;border-radius:50%;background:#10b981;display:inline-block;"></span>Online: <strong>' +
    online +
    "</strong></div>" +
    '<div style="display:flex;align-items:center;gap:8px;"><span style="width:10px;height:10px;border-radius:50%;background:#ef4444;display:inline-block;"></span>Offline: <strong>' +
    offline +
    "</strong></div>" +
    "</div></div>"
  );
}

// HELPER: TANGGAL
// ============================================================
function getDateRange(type, value) {
  var now = new Date();
  switch (type) {
    case "daily": {
      var d = value ? new Date(value) : now;
      var start = new Date(d);
      start.setHours(0, 0, 0, 0);
      var end = new Date(d);
      end.setHours(23, 59, 59, 999);
      return { start: start, end: end };
    }
    case "weekly": {
      var d = value ? new Date(value) : now;
      var day = d.getDay();
      var diff = day === 0 ? -6 : 1 - day;
      var start = new Date(d);
      start.setDate(d.getDate() + diff);
      start.setHours(0, 0, 0, 0);
      var end = new Date(start);
      end.setDate(start.getDate() + 6);
      end.setHours(23, 59, 59, 999);
      return { start: start, end: end };
    }
    case "monthly": {
      var base = value ? new Date(value + "-01") : now;
      var start = new Date(base.getFullYear(), base.getMonth(), 1);
      var end = new Date(base.getFullYear(), base.getMonth() + 1, 0);
      end.setHours(23, 59, 59, 99);
      return { start: start, end: end };
    }
    default:
      return null;
  }
}

function formatSqlDate(date) {
  return date.toISOString().slice(0, 19).replace("T", " ");
}

// PDF RENDERER
// ============================================================
async function renderPdf(html) {
  console.log("📄 Rendering PDF... HTML: " + (html.length / 1024).toFixed(1) + " KB");

  var browser = await puppeteer.launch({
    headless: "new",
    args: ["--no-sandbox", "--disable-setuid-sandbox", "--disable-dev-shm-usage", "--disable-gpu"],
  });

  try {
    var page = await browser.newPage();
    await page.setDefaultNavigationTimeout(0);
    await page.setDefaultTimeout(0);
    await page.setViewport({ width: A4_WIDTH_PX, height: A4_HEIGHT_PX, deviceScaleFactor: 1 });
    await page.setContent(html, { waitUntil: "domcontentloaded", timeout: 0 });
    await new Promise((resolve) => setTimeout(resolve, 800));

    var pdfBuffer = await page.pdf({
      format: "A4",
      printBackground: true,
      margin: { top: "0", bottom: "0", left: "0", right: "0" },
    });

    console.log("✅ PDF: " + (pdfBuffer.length / 1024).toFixed(1) + " KB");
    return pdfBuffer;
  } finally {
    await browser.close();
  }
}

async function generateReport(params) {
  var startDate = params.startDate;
  var endDate = params.endDate;
  var title = params.title;
  var subtitle = params.subtitle;
  var filename = params.filename;
  var res = params.res;

  try {
    var queries = await Promise.all([
      pool.query(
        "SELECT jenis_kendaraan, COUNT(*) as total FROM vehicle_log WHERE timestamp >= $1 AND timestamp <= $2 GROUP BY jenis_kendaraan",
        [startDate, endDate],
      ),
      pool.query(
        "SELECT COUNT(*) as total FROM alert_log WHERE LOWER(class_name) = 'box' AND created_at >= $1 AND created_at <= $2",
        [startDate, endDate],
      ),
      pool.query(
        "SELECT COUNT(*) as total FROM alert_log WHERE class_name = 'KAMERA OFFLINE' AND created_at >= $1 AND created_at <= $2",
        [startDate, endDate],
      ),
      pool.query(
        "SELECT timestamp, jenis_kendaraan, warna, confidence, gambar_base64, status_muatan FROM vehicle_log WHERE timestamp >= $1 AND timestamp <= $2 ORDER BY timestamp DESC LIMIT 100",
        [startDate, endDate],
      ),
      pool.query(
        "SELECT created_at, camera, class_name, foto_base64 FROM alert_log WHERE LOWER(class_name) = 'box' AND created_at >= $1 AND created_at <= $2 ORDER BY created_at DESC LIMIT 100",
        [startDate, endDate],
      ),
      pool.query(
        "SELECT camera, created_at FROM alert_log WHERE class_name = 'KAMERA OFFLINE' AND created_at >= $1 AND created_at <= $2 ORDER BY created_at DESC LIMIT 100",
        [startDate, endDate],
      ),
      pool.query(
        "SELECT EXTRACT(HOUR FROM timestamp) as hour, COUNT(*) as total FROM vehicle_log WHERE timestamp >= $1 AND timestamp <= $2 GROUP BY hour ORDER BY hour",
        [startDate, endDate],
      ),
      pool.query(
        "SELECT TO_CHAR(created_at, 'Dy') AS day, COUNT(*) AS total FROM alert_log WHERE created_at >= NOW() - INTERVAL '7 days' AND LOWER(class_name) = 'box' GROUP BY TO_CHAR(created_at, 'Dy'), DATE_TRUNC('day', created_at) ORDER BY DATE_TRUNC('day', created_at)",
      ),
      pool.query(
        'SELECT e."timestamp", e.event_type, e.confidence, emp.name AS employee_name, emp.employee_id AS nik FROM attendance_event e LEFT JOIN employees emp ON e.employee_id = emp.employee_id WHERE e."timestamp" >= $1 AND e."timestamp" <= $2 ORDER BY e."timestamp" DESC LIMIT 100',
        [startDate, endDate],
      ),
    ]);

    var vehicleStat = queries[0];
    var stagingStat = queries[1];
    var cameraStat = queries[2];
    var vehiclePhotos = queries[3];
    var stagingPhotos = queries[4];
    var cameraOffline = queries[5];
    var hourlyVehicles = queries[6];
    var stagingWeekly = queries[7];
    var attendanceData = queries[8];

    // Total kendaraan
    var totalKendaraan = 0;
    vehicleStat.rows.forEach(function (v) {
      totalKendaraan += Number(v.total);
    });

    // Attendance stats
    var totalArrival = 0;
    attendanceData.rows.forEach(function (a) {
      if (a.event_type === "ARRIVAL") totalArrival++;
    });

    // HITUNG PELANGGARAN JAM KERJA
    var totalPelanggaran = hitungPelanggaran(attendanceData.rows);

    var totalStaging = Number(stagingStat.rows[0]?.total || 0);
    var totalCameraOff = Number(cameraStat.rows[0]?.total || 0);
    var DEFAULT_TOTAL_CAMERAS = 22;
    var totalOnline = DEFAULT_TOTAL_CAMERAS - totalCameraOff;
    var maxWeekly = stagingWeekly.rows.reduce(function (max, r) {
      return Math.max(max, Number(r.total));
    }, 0);
    var maxHourly = hourlyVehicles.rows.reduce(function (max, r) {
      return Math.max(max, Number(r.total));
    }, 0);

    // KPI CARDS
    var kpiCards =
      '<div class="kpi-row">' +
      '<div class="kpi-card c1"><div class="kpi-icon" style="background:linear-gradient(135deg,#6366f1,#818cf8);">🚛</div><div class="kpi-val">' +
      totalKendaraan +
      '</div><div class="kpi-lbl">Total Kendaraan</div><div class="kpi-sub">Mobil+Truk+Motor</div></div>' +
      '<div class="kpi-card c2"><div class="kpi-icon" style="background:linear-gradient(135deg,#10b981,#34d399);">✅</div><div class="kpi-val">' +
      totalArrival +
      '</div><div class="kpi-lbl">Kehadiran</div><div class="kpi-sub">Karyawan Hadir</div></div>' +
      '<div class="kpi-card c3"><div class="kpi-icon" style="background:linear-gradient(135deg,#ef4444,#f87171);">⚠️</div><div class="kpi-val">' +
      totalPelanggaran +
      '</div><div class="kpi-lbl">Pelanggaran</div><div class="kpi-sub">Jam Kerja</div></div>' +
      '<div class="kpi-card c4"><div class="kpi-icon" style="background:linear-gradient(135deg,#8b5cf6,#a78bfa);">📦</div><div class="kpi-val">' +
      totalStaging +
      '</div><div class="kpi-lbl">Staging</div><div class="kpi-sub">Box Terdeteksi</div></div>' +
      '<div class="kpi-card c5"><div class="kpi-icon" style="background:linear-gradient(135deg,#f59e0b,#fbbf24);">📡</div><div class="kpi-val">' +
      totalOnline +
      "/" +
      DEFAULT_TOTAL_CAMERAS +
      '</div><div class="kpi-lbl">Camera Online</div><div class="kpi-sub">' +
      totalCameraOff +
      " Offline</div></div>" +
      "</div>";

    // Charts
    var hourlyData = hourlyVehicles.rows.map(function (r) {
      return { label: String(r.hour).padStart(2, "0") + ":00", total: Number(r.total) };
    });
    var stagingData = stagingWeekly.rows.map(function (r) {
      return { label: r.day, total: Number(r.total) };
    });
    var chartHours = generateBarChart(hourlyData, "label", "total", maxHourly, "#6366f1");
    var chartStaging = generateBarChart(stagingData, "label", "total", maxWeekly, "#8b5cf6");
    var donutChart = generateDonutChart(DEFAULT_TOTAL_CAMERAS, totalCameraOff);

    var chartsRow =
      '<div class="chart-row">' +
      '<div class="card"><div class="card-hd">⏰ Vehicles per Hour</div>' +
      chartHours +
      "</div>" +
      '<div class="card"><div class="card-hd">📊 Staging History (7 Days)</div>' +
      chartStaging +
      "</div>" +
      "</div>" +
      '<div class="chart-row">' +
      '<div class="card"><div class="card-hd">📡 Camera Status</div>' +
      donutChart +
      "</div>" +
      '<div class="card"><div class="card-hd">📋 Activity Summary</div>' +
      '<div style="display:flex;flex-direction:column;gap:8px;font-size:11px;padding:8px 0;">' +
      '<div style="display:flex;justify-content:space-between;padding:6px 0;border-bottom:1px solid #f1f5f9;"><span>🚛 Total Kendaraan</span><strong>' +
      totalKendaraan +
      "</strong></div>" +
      '<div style="display:flex;justify-content:space-between;padding:6px 0;border-bottom:1px solid #f1f5f9;"><span>✅ Kehadiran</span><strong style="color:#10b981;">' +
      totalArrival +
      "</strong></div>" +
      '<div style="display:flex;justify-content:space-between;padding:6px 0;border-bottom:1px solid #f1f5f9;"><span>⚠️ Pelanggaran Jam Kerja</span><strong style="color:#ef4444;">' +
      totalPelanggaran +
      "</strong></div>" +
      '<div style="display:flex;justify-content:space-between;padding:6px 0;border-bottom:1px solid #f1f5f9;"><span>📦 Staging</span><strong style="color:#8b5cf6;">' +
      totalStaging +
      "</strong></div>" +
      '<div style="display:flex;justify-content:space-between;padding:6px 0;"><span>📡 Camera Online</span><strong style="color:#f59e0b;">' +
      totalOnline +
      "/" +
      DEFAULT_TOTAL_CAMERAS +
      "</strong></div>" +
      "</div></div></div>";

    // Camera, Attendance, Vehicle, Staging sections (sama seperti sebelumnya)
    var cameraSection =
      cameraOffline.rows.length > 0
        ? '<div class="tbl-wrap"><table><thead><tr><th>No</th><th>Camera</th><th>Time</th></tr></thead><tbody>' +
          cameraOffline.rows
            .map(function (c, i) {
              return (
                "<tr><td><strong>" +
                (i + 1) +
                "</strong></td><td>🔴 " +
                c.camera +
                "</td><td>" +
                new Date(c.created_at).toLocaleString("id-ID", {
                  dateStyle: "short",
                  timeStyle: "short",
                }) +
                "</td></tr>"
              );
            })
            .join("") +
          "</tbody></table></div>"
        : '<div class="empty">✅ All cameras normal</div>';

    var attendanceSection =
      attendanceData.rows.length > 0
        ? '<div class="tbl-wrap"><table><thead><tr><th>No</th><th>Name</th><th>NIK</th><th>Status</th><th>Time</th><th>Conf</th></tr></thead><tbody>' +
          attendanceData.rows
            .map(function (a, i) {
              var badge =
                a.event_type === "ARRIVAL"
                  ? "badge-g"
                  : a.event_type === "DEPARTURE"
                    ? "badge-y"
                    : a.event_type === "BREAK_OUT"
                      ? "badge-r"
                      : "badge-b";
              var label =
                a.event_type === "ARRIVAL"
                  ? "Masuk"
                  : a.event_type === "DEPARTURE"
                    ? "Pulang"
                    : a.event_type === "BREAK_OUT"
                      ? "Istirahat"
                      : "Kembali";
              return (
                "<tr><td>" +
                (i + 1) +
                "</td><td><strong>" +
                (a.employee_name || "Unknown") +
                "</strong></td><td>" +
                (a.nik || "-") +
                '</td><td><span class="badge ' +
                badge +
                '">' +
                label +
                "</span></td><td>" +
                new Date(a.timestamp).toLocaleString("id-ID", {
                  dateStyle: "short",
                  timeStyle: "short",
                }) +
                "</td><td>" +
                (a.confidence ? (a.confidence * 100).toFixed(1) + "%" : "-") +
                "</td></tr>"
              );
            })
            .join("") +
          "</tbody></table></div>"
        : '<div class="empty">📋 No attendance records</div>';

    var vehicleSection =
      vehiclePhotos.rows.length > 0
        ? '<div class="photo-grid">' +
          vehiclePhotos.rows
            .map(function (v) {
              var img = resizeBase64Image(v.gambar_base64);
              return (
                '<div class="photo-card"><div class="photo-img">' +
                (img
                  ? '<img src="data:image/jpeg;base64,' + img + '" />'
                  : '<div class="photo-noimg">📷 No Image</div>') +
                '</div><div class="photo-info"><div class="photo-title">🚛 ' +
                (v.jenis_kendaraan || "Vehicle") +
                '</div><div class="photo-detail"><span class="photo-lbl">Color</span> ' +
                (v.warna || "N/A") +
                "</div>" +
                (v.status_muatan
                  ? '<div class="photo-detail"><span class="photo-lbl">Load</span> ' +
                    v.status_muatan +
                    "</div>"
                  : "") +
                '<div class="photo-detail"><span class="photo-lbl">Time</span> ' +
                new Date(v.timestamp).toLocaleString("id-ID", {
                  dateStyle: "short",
                  timeStyle: "short",
                }) +
                '</div><div class="photo-conf">✓ ' +
                (v.confidence ? (v.confidence * 100).toFixed(1) : "0") +
                "%</div></div></div>"
              );
            })
            .join("") +
          "</div>"
        : '<div class="empty">🚛 No vehicle records</div>';

    var stagingSection =
      stagingPhotos.rows.length > 0
        ? '<div class="photo-grid">' +
          stagingPhotos.rows
            .map(function (s) {
              var img = resizeBase64Image(s.foto_base64);
              return (
                '<div class="photo-card"><div class="photo-img">' +
                (img
                  ? '<img src="data:image/jpeg;base64,' + img + '" />'
                  : '<div class="photo-noimg">📷 No Image</div>') +
                '</div><div class="photo-info"><div class="photo-title">📦 ' +
                (s.class_name || "Box") +
                '</div><div class="photo-detail"><span class="photo-lbl">Camera</span> ' +
                (s.camera || "N/A") +
                '</div><div class="photo-detail"><span class="photo-lbl">Time</span> ' +
                new Date(s.created_at).toLocaleString("id-ID", {
                  dateStyle: "short",
                  timeStyle: "short",
                }) +
                "</div></div></div>"
              );
            })
            .join("") +
          "</div>"
        : '<div class="empty">📦 No staging records</div>';

    var html = buildMainHTML({
      title: title,
      subtitle: subtitle,
      logoBase64: logoBase64,
      kpiCards: kpiCards,
      chartsRow: chartsRow,
      cameraSection: cameraSection,
      attendanceSection: attendanceSection,
      vehicleSection: vehicleSection,
      stagingSection: stagingSection,
    });

    console.log("📏 HTML: " + (html.length / 1024).toFixed(1) + " KB");
    var pdfBuffer = await renderPdf(html);
    res.setHeader("Content-Type", "application/pdf");
    res.setHeader("Content-Disposition", 'attachment; filename="' + filename + '"');
    res.send(pdfBuffer);
  } catch (err) {
    console.error("Generate report error:", err);
    res.status(500).json({ error: "Gagal generate laporan" });
  }
}

    res.setHeader("Content-Disposition", 'attachment; filename="monitoring_report.pdf"');
    res.send(pdfBuffer);
  } catch (err) {
    console.error("Full report error:", err);
    res.status(500).json({ error: "Gagal generate laporan" });
  }
});

router.get("/daily", async function (req, res) {
  var date = req.query.date;
  var startDate, endDate;
  if (date) {
    startDate = date;
    endDate = date + " 23:59:59";
  } else {
    var today = new Date();
    startDate =
      today.getFullYear() +
      "-" +
      String(today.getMonth() + 1).padStart(2, "0") +
      "-" +
      String(today.getDate()).padStart(2, "0");
    endDate = startDate + " 23:59:59";
  }
  await generateReport({
    startDate: startDate,
    endDate: endDate,
    title: "Daily Report",
    subtitle: new Date(startDate).toLocaleDateString("id-ID", {
      weekday: "long",
      day: "numeric",
      month: "long",
      year: "numeric",
    }),
    filename: "daily_" + startDate + ".pdf",
    res: res,
  });
});

router.get("/report", async function (req, res) {
  var type = req.query.type || "daily";
  if (type === "full") return res.redirect("/api/export/full-report");
  if (type === "daily") return res.redirect("/api/export/daily?date=" + (req.query.date || ""));
  if (type === "weekly") return res.redirect("/api/export/weekly?date=" + (req.query.date || ""));
  if (type === "monthly")
    return res.redirect("/api/export/monthly?month=" + (req.query.month || ""));
  res.status(400).json({ error: "Invalid report type" });
});

router.get("/attendance/log", async function (req, res) {
  try {
    var conditions = [],
      params = [];
    if (req.query.start_date) {
      params.push(req.query.start_date);
      conditions.push('e."timestamp" >= $' + params.length);
    }
    if (req.query.end_date) {
      params.push(req.query.end_date);
      conditions.push('e."timestamp" <= $' + params.length);
    }
    var whereClause = conditions.length > 0 ? "WHERE " + conditions.join(" AND ") : "";
    var query =
      'SELECT e.id, e."timestamp", e.event_type, e.direction, e.confidence, e.camera_id, e.is_visitor, e.snapshot_path, emp.name AS employee_name, emp.employee_id AS nik FROM attendance_event e LEFT JOIN employees emp ON e.employee_id = emp.employee_id ' +
      whereClause +
      ' ORDER BY e."timestamp" DESC LIMIT $' +
      (params.length + 1);
    var result = await pool.query(query, params.concat([req.query.limit || 20]));
    res.json({ data: result.rows, total: result.rowCount });
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: "Failed" });
  }
});

router.get("/attendance/stats/daily", async function (req, res) {
  try {
    var query =
      "WITH daily_counts AS (SELECT event_type, COUNT(*) as total FROM attendance_event WHERE \"timestamp\"::date = CURRENT_DATE GROUP BY event_type) SELECT COALESCE(SUM(CASE WHEN event_type = 'ARRIVAL' THEN total ELSE 0 END), 0) as total_arrival, COALESCE(SUM(CASE WHEN event_type = 'DEPARTURE' THEN total ELSE 0 END), 0) as total_departure, COALESCE(SUM(CASE WHEN event_type = 'BREAK_OUT' THEN total ELSE 0 END), 0) as total_break_out, COALESCE(SUM(CASE WHEN event_type = 'BREAK_IN' THEN total ELSE 0 END), 0) as total_break_in FROM daily_counts";
    var result = await pool.query(query);
    res.json(result.rows[0]);
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});
